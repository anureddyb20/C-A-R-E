PROJECT: ACCESSIBLE WEARABLE HEALTH & STRESS MONITOR

------------------------------------------------------------------------
1. PROJECT SUMMARY & SIGNIFICANCE
------------------------------------------------------------------------
Our project addresses the critical lack of affordable, continuous health monitoring for the elderly and patients in low-resource areas. Currently, continuous cardiac and stress monitoring requires expensive, medical-grade equipment or premium smartwatches that mandate monthly subscriptions and rely on complex smartphone applications. As a result, sudden cardiac events or panic attacks often go unnoticed until it is too late, leaving vulnerable individuals at severe risk. 

By utilizing ultra-low-cost components (~₹1,100), our project democratizes access to life-saving technology. It provides a standalone monitoring system requiring zero technical literacy. If an abnormal heart rate is detected, or the panic button is triggered, caregivers receive instant, localized alerts on a dedicated receiver, drastically reducing emergency response times and providing peace of mind.

------------------------------------------------------------------------
2. PHYSICAL COMPONENTS & ARCHITECTURE
------------------------------------------------------------------------
The system consists of a wearable transmitter and a stationary caregiver receiver. The wearable node is powered by an STM32F103C8T6 microcontroller, chosen for its fast processing and highly accurate 12-bit ADCs. It utilizes an AD8232 module for single-lead ECG tracking and a custom Galvanic Skin Response (GSR) sensor with metal electrodes to measure stress-induced sweat activity. 

For telemetry, an nRF24L01+ radio module transmits biometric data locally with zero latency. An onboard 3V vibration motor, safely isolated via a 2N2222 transistor, provides immediate haptic feedback to the patient. The device runs on a 3.7V LiPo battery managed by a TP4056 charging module. 

Structurally, the circuitry is housed in a compact, 3D-printed enclosure worn on the forearm. The GSR electrodes maintain skin contact via an elastic strap, while the tactile panic button on the top face allows users to easily trigger a manual alarm during emergencies.

------------------------------------------------------------------------
3. CIRCUIT DIAGRAM & CONNECTIONS
------------------------------------------------------------------------
STM32 Blue Pill Pinout Plan:
- Power:
  - TP4056 OUT+ -> STM32 5V pin
  - TP4056 OUT- -> STM32 GND pin
- AD8232 (ECG):
  - 3.3V -> 3.3V
  - GND -> GND
  - OUTPUT -> PA0 (ADC)
  - LO+ -> PB0 (Digital In)
  - LO- -> PB1 (Digital In)
- GSR Sensor:
  - Divider output -> PA1 (ADC) (Using a 10k resistor to GND)
- nRF24L01+ (SPI1):
  - CE -> PA4
  - CSN -> PA3
  - SCK -> PA5
  - MISO -> PA6
  - MOSI -> PA7
- Vibration Motor (via 2N2222 NPN Transistor):
  - Base -> PB10 (via 1k resistor)
  - Emitter -> GND
  - Collector -> Motor Negative
  - Motor Positive -> 3.3V/5V
- Push Button (Panic):
  - Button -> PB12 to GND (with internal pull-up)

4. WEARABLE TRANSMITTER CODE (STM32 / wearable_tx.ino)
------------------------------------------------------------------------
// Wearable Transmitter (STM32 Blue Pill)
#include <SPI.h>
#include <nRF24L01.h>
#include <RF24.h>

// --- PIN DEFINITIONS ---
#define ECG_PIN PA0       // AD8232 Output
#define GSR_PIN PA1       // GSR Voltage Divider
#define LO_PLUS PB0       // AD8232 Leads Off +
#define LO_MINUS PB1      // AD8232 Leads Off -
#define MOTOR_PIN PB10    // Vibration Motor (via 2N2222)
#define BUTTON_PIN PB12   // Panic Button (Active Low)

// nRF24L01 Pins (SPI1)
#define CE_PIN PA4
#define CSN_PIN PA3

RF24 radio(CE_PIN, CSN_PIN);
const byte address[6] = "00001";

// --- DATA STRUCTURE ---
struct SensorData {
  uint16_t ecg_value;
  uint8_t heart_rate;
  uint16_t gsr_value;
  bool panic_button;
  bool leads_off;
};

SensorData payload;

// --- HR CALCULATION VARIABLES ---
unsigned long lastBeatTime = 0;
int threshold = 3000; // Adjust based on ADC (12-bit STM32 is 0-4095)
bool peakDetected = false;
unsigned long lastSampleTime = 0;

void setup() {
  Serial.begin(115200);
  
  pinMode(ECG_PIN, INPUT_ANALOG);
  pinMode(GSR_PIN, INPUT_ANALOG);
  pinMode(LO_PLUS, INPUT);
  pinMode(LO_MINUS, INPUT);
  
  pinMode(MOTOR_PIN, OUTPUT);
  pinMode(BUTTON_PIN, INPUT_PULLUP);
  digitalWrite(MOTOR_PIN, LOW);
  
  // Initialize Radio
  if (!radio.begin()) {
    Serial.println("Radio hardware not responding!");
    while (1) {} // Hold
  }
  radio.openWritingPipe(address);
  radio.setPALevel(RF24_PA_MAX);
  radio.stopListening(); // Transmitter mode
}

void loop() {
  unsigned long currentTime = millis();
  
  // Sample at ~100Hz (every 10ms)
  if (currentTime - lastSampleTime >= 10) {
    lastSampleTime = currentTime;
    
    // Read Sensors
    payload.ecg_value = analogRead(ECG_PIN);
    payload.gsr_value = analogRead(GSR_PIN);
    payload.panic_button = (digitalRead(BUTTON_PIN) == LOW);
    payload.leads_off = (digitalRead(LO_PLUS) == HIGH || digitalRead(LO_MINUS) == HIGH);
    
    // Basic Peak Detection for Heart Rate
    if (payload.ecg_value > threshold && !peakDetected) {
      peakDetected = true;
      unsigned long beatInterval = currentTime - lastBeatTime;
      lastBeatTime = currentTime;
      
      if (beatInterval > 300 && beatInterval < 2000) { // Valid interval (30-200 BPM)
        payload.heart_rate = 60000 / beatInterval;
      }
    } else if (payload.ecg_value < threshold - 200) {
      peakDetected = false;
    }
    
    // Alert System: Trigger Motor
    if (payload.panic_button || (payload.heart_rate > 120) || (payload.heart_rate > 0 && payload.heart_rate < 50)) {
      digitalWrite(MOTOR_PIN, HIGH);
    } else {
      digitalWrite(MOTOR_PIN, LOW);
    }
    
    // Transmit
    radio.write(&payload, sizeof(SensorData));
  }
}

5. GATEWAY RECEIVER CODE (ESP32 / gateway_rx.ino)

// Gateway Receiver (ESP32)
#include <WiFi.h>
#include <WebSocketsServer.h> // Requires WebSockets by Markus Sattler
#include <SPI.h>
#include <nRF24L01.h>
#include <RF24.h>

// --- WIFI SETTINGS ---
const char* ssid = "YOUR_WIFI_SSID";
const char* password = "YOUR_WIFI_PASSWORD";

// --- NRF24 PINS ---
#define CE_PIN 4
#define CSN_PIN 5
// Default ESP32 SPI: SCK=18, MISO=19, MOSI=23

RF24 radio(CE_PIN, CSN_PIN);
const byte address[6] = "00001";

// --- WEBSOCKET SERVER ---
WebSocketsServer webSocket = WebSocketsServer(81);

struct SensorData {
  uint16_t ecg_value;
  uint8_t heart_rate;
  uint16_t gsr_value;
  bool panic_button;
  bool leads_off;
};

SensorData payload;
unsigned long lastSend = 0;

void setup() {
  Serial.begin(115200);
  
  // Connect to WiFi
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi connected. IP: ");
  Serial.println(WiFi.localIP());
  
  // Start WebSockets
  webSocket.begin();
  
  // Initialize Radio
  if (!radio.begin()) {
    Serial.println("Radio hardware not responding!");
    while (1) {}
  }
  radio.openReadingPipe(0, address);
  radio.setPALevel(RF24_PA_MAX);
  radio.startListening();
}

void loop() {
  webSocket.loop();
  
  if (radio.available()) {
    radio.read(&payload, sizeof(SensorData));
    
    // Limit websocket broadcast to ~20Hz to prevent browser overload
    if (millis() - lastSend > 50) {
      lastSend = millis();
      
      // Format as JSON
      String json = "{";
      json += "\"ecg\":" + String(payload.ecg_value) + ",";
      json += "\"bpm\":" + String(payload.heart_rate) + ",";
      json += "\"gsr\":" + String(payload.gsr_value) + ",";
      json += "\"panic\":" + String(payload.panic_button ? "true" : "false") + ",";
      json += "\"leads_off\":" + String(payload.leads_off ? "true" : "false");
      json += "}";
      
      webSocket.broadcastTXT(json);
    }
  }
}

        let ws;
        
        // Chart Setup
        const ctx = document.getElementById('ecgChart').getContext('2d');
        const maxDataPoints = 100;
        
        const ecgChart = new Chart(ctx, {
            type: 'line',
            data: {
                labels: Array(maxDataPoints).fill(''),
                datasets: [{
                    label: 'ECG Trace',
                    data: Array(maxDataPoints).fill(0),
                    borderColor: '#4ade80',
                    borderWidth: 2,
                    pointRadius: 0,
                    tension: 0.1

        function connect() {
            const ip = document.getElementById('ipInput').value;
            if(!ip) return;
            
            if(ws) ws.close();
            
            document.getElementById('connection').innerText = `Connecting to ws://${ip}:81...`;
            ws = new WebSocket(`ws://${ip}:81`);

            ws.onopen = () => {
                document.getElementById('connection').innerText = "🟢 Connected";
                document.getElementById('connection').style.color = "#4ade80";
            };

            ws.onclose = () => {
                document.getElementById('connection').innerText = "🔴 Disconnected";
                document.getElementById('connection').style.color = "#ef4444";
            };

            ws.onmessage = (event) => {
                const data = JSON.parse(event.data);
                
                // Update text
                document.getElementById('bpmValue').innerText = data.bpm;
                document.getElementById('gsrValue').innerText = data.gsr;
                
                // Status Logic
                const statusCard = document.getElementById('statusCard');
                const statusValue = document.getElementById('statusValue');
                
                if (data.panic) {
                    statusValue.innerText = "PANIC BUTTON!";
                    statusValue.className = "value alert";
                    statusCard.style.border = "2px solid #ef4444";
                } else if (data.leads_off) {
                    statusValue.innerText = "Check Sensors";
                    statusValue.className = "value";
                    statusValue.style.color = "#fbbf24";
                    statusCard.style.border = "1px solid #333";
                } else if (data.bpm > 120 || (data.bpm > 0 && data.bpm < 50)) {
                    statusValue.innerText = "HR ALERT";
                    statusValue.className = "value alert";
                    statusCard.style.border = "2px solid #ef4444";
                } else {
                    statusValue.innerText = "Normal";
                    statusValue.className = "value";
                    statusValue.style.color = "#4ade80";
                    statusCard.style.border = "1px solid #333";
                }

                // Update Chart
                let chartData = ecgChart.data.datasets[0].data;
                chartData.push(data.ecg);
                if (chartData.length > maxDataPoints) chartData.shift();
                ecgChart.update();
            };
        }
